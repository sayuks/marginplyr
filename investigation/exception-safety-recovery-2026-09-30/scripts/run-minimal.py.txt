from pathlib import Path
import subprocess,os,json
root=Path(__file__).resolve().parent.parent
for mode in ['error','interrupt']:
 for rep in range(1,4):
  out=root/'results'/f'minimal--{mode}--{rep}';out.mkdir()
  env=os.environ.copy();env.update(HOME=str(root/'home'),TMPDIR=str(root/'tmp'),R_LIBS_USER=str(root/'library'),R_LIBS_SITE='')
  p=subprocess.run(['Rscript','--vanilla',str(root/'scripts/minimal.R'),str(root),mode,str(out)],env=env,capture_output=True,text=True,timeout=60)
  (out/'console.log').write_text(p.stdout+p.stderr)
  if (out/'minimal.json').exists():
   x=json.loads((out/'minimal.json').read_text());print(mode,rep,x['b'],x['next_transaction'],x['retry'],x['independent'])
  else:print(p.returncode,p.stderr)
  for f in out.glob('*.sqlite*'):f.unlink()
