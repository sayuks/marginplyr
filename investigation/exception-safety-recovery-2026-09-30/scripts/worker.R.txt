snapshot <- function(passive=FALSE) {
  handles<-lapply(state$handles,function(h) tryCatch(DBI::dbIsValid(h),error=function(e)NA))
  valid<-DBI::dbIsValid(con)
  result<-list(valid=valid,query_sql_unchanged=if(valid)identical(q_render,as.character(dbplyr::sql_render(q)))else NA,handles_valid=handles,input_structure_unchanged=identical(source_signature,serialize(list(class=class(source),query=source$lazy_query),NULL)),query_structure_unchanged=identical(q_signature,serialize(list(class=class(q),query=q$lazy_query),NULL)),options=opts(),input_object_unchanged=identical(serialize(source,NULL),sourcecopy),query_object_unchanged=identical(serialize(q,NULL),qcopy),audit=capture(last_sent_queries()),cache=as.list(get("share_dialect_verdicts",ns),all.names=TRUE))
  if(!valid || passive || (startsWith(case,"collect_") && any(unlist(handles),na.rm=TRUE)))return(result)
  result$schemas<-lapply(c("main","temp",if(variant=="attached")"other"),function(s) {
    catalog<-DBI::dbGetQuery(con,paste0("SELECT type,name,tbl_name,sql FROM ",s,".sqlite_master ORDER BY type,name"))
    tables<-setNames(lapply(catalog$name[catalog$type=="table"],function(t) {
      id<-DBI::dbQuoteIdentifier(con,DBI::Id(schema=s,table=t));DBI::dbGetQuery(con,paste0("SELECT * FROM ",id))}),catalog$name[catalog$type=="table"])
    list(schema=s,catalog=catalog,tables=tables)
  });result
}

fault <- function(label) {
  state$hit<-state$hit+1L;state$active<-FALSE
  json(list(case=case,mode=mode,boundary=label,pid=Sys.getpid(),hit=state$hit,side_effects=snapshot(passive=startsWith(case,"collect_"))),file.path(outdir,"reached.json"))
  if(mode=="error")stop(structure(list(message=paste("fault",case,label),call=NULL),class=c("fault_injected","error","condition")))
  if(mode=="interrupt")rlang::interrupt()
  if(mode=="sigint") {writeLines("ready",file.path(outdir,"ready"));repeat Sys.sleep(0.05)}
  invisible(NULL)
}
hook <- function(stage, connection, statement, value=NULL) {
 if(!identical(connection,con))return(invisible(NULL))
 sql<-as.character(statement);success<-stage=="after" && !is.null(value)
 state$events[[length(state$events)+1L]]<-list(stage=stage,sql=sql,success=success)
 if(grepl("^SAVEPOINT",sql))state$savepoint<-sub('^SAVEPOINT ',"",sql)
 if(!state$active)return(invisible(NULL))
 target <- FALSE
 if(case=="begin_before")target<-stage=="before" && grepl("^SAVEPOINT",sql)
 if(case=="begin_after")target<-success && grepl("^SAVEPOINT",sql)
 if(case=="drop_after")target<-success && grepl("^DROP TABLE",sql) && grepl("report",sql)
 if(case=="create_after")target<-success && grepl("^CREATE.*TABLE",sql) && grepl("report",sql)
 if(case=="index_after")target<-success && grepl("^CREATE.*INDEX",sql) && grepl("report",sql)
 if(case %in% c("insert_after","cleanup_between","invalid_cleanup"))target<-success && grepl("^INSERT INTO",sql) && grepl("report",sql)
 if(case=="analyze_after")target<-success && grepl("^ANALYZE",sql)
 if(case=="release_after")target<-success && grepl("^RELEASE SAVEPOINT",sql)
 if(success && grepl("^INSERT INTO",sql) && grepl("report",sql))state$inserted<-TRUE
 if(case=="cleanup_between") {
   if(target && !state$primary){state$primary<-TRUE;stop(structure(list(message="primary failure after insert",call=NULL),class=c("fault_injected","error","condition")))}
   target<-state$primary && success && grepl("^ROLLBACK TO",sql)
 }
 if(case=="invalid_cleanup" && target) {state$active<-FALSE;json(snapshot(),file.path(outdir,"before-invalidating.json"));DBI::dbDisconnect(con);stop("primary failure with invalidated connection")}
 if(target)fault(paste(stage,sql))
 invisible(NULL)
}
